__device__ float sin32Pow(float sinSq) { return pow16(sinSq); } //RR!!

__device__ bool tryForwardVote(float3 radial, RdvCache* cache, int* cacheCount) {
    // Volle scan, orthogonale sin32 only
    float bestWeight = minVoteWeight;
    int bestIdx = -1;
    for (int i = 0; i < *cacheCount; ++i) {
        float dot = fabsf(dot(radial, cache->entries[i].axis));
        if (dot < cos75) {
            float sinSq = 1.0f - dot * dot;
            float weight = sin32Pow(sinSq); //RR!!
            if (weight > bestWeight) {
                bestWeight = weight;
                bestIdx = i;
            }
        }
    }
    if (bestIdx != -1) {
        atomicAdd(&cache->entries[bestIdx].voteWeight, bestWeight);
        // Bubble up
        int j = bestIdx;
        while (j > 0 && cache->entries[j].voteWeight > cache->entries[j - 1].voteWeight) {
            swapEntries(cache->entries[j], cache->entries[j - 1]);
            --j;
        }
        // Parabool nudge
        CacheEntry* entry = &cache->entries[bestIdx];
        atomicAdd(&entry->voteCount, 1U);
        u32 count = entry->voteCount;
        float countF = (float)count;
        float alpha = 1.0f + paraboolA * countF * countF;
        if (alpha > alphaStake) {
            float3 nudge = (1.0f - alpha) * entry->axis + alpha * normalize(radial) * bestWeight;
            entry->axis = normalize(nudge);
        }
        return true;
    }
    return false;
}

// Onze Boeren Ring impl (streaming, 4-pt test, sliding chunks)
#include "k_generate_normals.cuh"
#include "config.h"
#include "rdv_kernel.cuh"  // processNormal

constexpr float nearPlane = 1e-3f;
constexpr float minAreaSq = 1e-6f;

__device__ bool passesPlanarityTest(const float3& p, const float3& p1, const float3& p2, const float3& p3, float3& normal) {
    // Triangle 1: p/p1/p2 test p3
    if (!isCollinear(p, p1, p2)) {
        float3 e1 = p1 - p;
        float3 e2 = p2 - p;
        normal = normalize(cross(e1, e2));
        float3 to_p3 = p3 - p;
        float distSq = dot(to_p3, normal) * dot(to_p3, normal);
        if (distSq < nearPlane * nearPlane) return true;
    }
    // Triangle 2: p1/p2/p3 test p
    if (!isCollinear(p1, p2, p3)) {
        float3 e1 = p2 - p1;
        float3 e2 = p3 - p1;
        normal = normalize(cross(e1, e2));
        float3 to_p = p - p1;
        float distSq = dot(to_p, normal) * dot(to_p, normal);
        if (distSq < nearPlane * nearPlane) return true;
    }
    return false;
}

__global__ void k_generate_normals_kernel(
    const float3* points, uint32_t chunkSize, u8* labels, bool directProcess
) {
    uint32_t i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= chunkSize) return;

    float3 p = points[i];
    int ringCount = 0;
    float3 ring[4];
    int ringIdx[4];
    int farPointCount = 0;
    int prevJ = i;

    for (int j = i + 1; j < chunkSize; ++j) {
        float3 neighbor = points[j];
        float distSq = lengthSq(neighbor - p);
        if (lengthSq(neighbor - points[prevJ]) > mortonLeapSq) {
            ++farPointCount;
            if (farPointCount >= 2) break;
        }
        prevJ = j;
        if (distSq <= gConfig.boerRing.r2Sq) {
            if (distSq > gConfig.boerRing.r1Sq && ringCount < 4) {
                ring[ringCount] = neighbor;
                ringIdx[ringCount] = j;
                ++ringCount;
            }
        }
    }

    if (ringCount < 3 || farPointCount >= 2) {
        labels[i] = PointType::chaos;
        for (int k = 0; k < ringCount; ++k) labels[ringIdx[k]] = PointType::chaos;
        return;
    }

    float3 normal;
    if (passesPlanarityTest(p, ring[0], ring[1], ring[2], normal)) {
        labels[i] = PointType::plane;
        if (directProcess) processNormal(normal, /* ring/cache */);
        return;
    }

    labels[i] = PointType::chaos;
    for (int k = 0; k < ringCount; ++k) labels[ringIdx[k]] = PointType::chaos;
}

#include "h_reduce_consensus.cuh"
#include "config.h"
#include <algorithm>
#include <vector>
#include <cmath>

constexpr float cos15 = 0.99f;  // cos(8.11°) voor chunk-merge

RdvCache combineResults(const std::vector<RdvCache>& chunkCaches) {
    RdvCache final;
    final.count = 0;

    for (const auto& chunk : chunkCaches) {
        for (int j = 0; j < chunk.count; ++j) {
            const CacheEntry& entry = chunk.entries[j];
            float bestDot = -1.0f;
            int bestIdx = -1;

            for (int k = 0; k < final.count; ++k) {
                float dot = fabsf(dot(entry.axis, final.entries[k].axis));
                if (dot > bestDot && dot > cos15) {  // Merge binnen 15°
                    bestDot = dot;
                    bestIdx = k;
                }
            }

            if (bestIdx != -1) {
                // Weighted average axes
                float totalW = final.entries[bestIdx].voteWeight + entry.voteWeight;
                float alpha = entry.voteWeight / totalW;
                final.entries[bestIdx].axis = normalize(
                    (1.0f - alpha) * final.entries[bestIdx].axis + alpha * entry.axis);
                final.entries[bestIdx].voteWeight = totalW;
                final.entries[bestIdx].voteCount += entry.voteCount;
            } else if (final.count < CACHE_SIZE) {
                final.entries[final.count++] = entry;
            }
        }
    }

    // Alternatief simpler voor kleine N (host, descending on key):
	std::vector<CacheEntry*> entry_ptrs(final.entries, final.entries + final.count);
	std::sort(entry_ptrs.begin(), entry_ptrs.end(), [](CacheEntry* a, CacheEntry* b) {
		return a->voteWeight > b->voteWeight;
	});
	for (int i = 0; i < final.count; ++i) {
		final.entries[i] = *entry_ptrs[i];
	}

    return final;
}

void h_reduce_global_consensus(const std::vector<std::vector<OTAsAccu>>& all_chunk_slots, AssenResultaat& final_result) {
    // Map OTAsAccu naar RdvCache (legacy compat)
    std::vector<RdvCache> caches;
    for (const auto& slots : all_chunk_slots) {
        RdvCache cache;
        // Copy logic: slots to entries, count from slots.size()
        caches.push_back(cache);
    }
    RdvCache merged = combineResults(caches);
    // Copy to final_result (v[0-2] = top axes)
    final_result.dimensie = merged.count;
    for (int i = 0; i < std::min(3, merged.count); ++i) {
        final_result.v[i][0] = merged.entries[i].axis.x;
        final_result.v[i][1] = merged.entries[i].axis.y;
        final_result.v[i][2] = merged.entries[i].axis.z;
    }
}
